@tool
def monte_carlo_suite(scenarios: List[ScenarioParams]) -> List[MonteCarloResult]:
    """Parallel Monte Carlo (10k paths/scenario)"""
    results = []
    for params in scenarios:
        paths = numpy_financial.monte_carlo_paths(
            initial=params.initial,
            mu=params.expected_return,
            sigma=params.volatility,
            n_periods=params.years * 12,
            n_paths=10000
        )
        results.append(MonteCarloResult(
            mean_return=np.mean(paths[-1]),
            p10=np.percentile(paths[-1], 10),
            p90=np.percentile(paths[-1], 90),
            sharpe=params.expected_return / params.volatility
        ))
    return results

def variation_generator(base_params: dict, user_risk: str) -> List[dict]:
    """Generate ±15% variations based on risk tolerance"""
    variations = []
    risk_adjust = {"conservative": -0.02, "moderate": 0.0, "aggressive": 0.02}
    for adj in [-0.15, 0.0, 0.15]:
        var = base_params.copy()
        var["expected_return"] += risk_adjust[user_risk] + adj
        variations.append(var)
    return variations

@tool
def monte_carlo_suite(scenarios: List[ScenarioParams]) -> List[MonteCarloResult]:
    """Parallel Monte Carlo (10k paths/scenario)"""
    results = []
    for params in scenarios:
        paths = numpy_financial.monte_carlo_paths(
            initial=params.initial,
            mu=params.expected_return,
            sigma=params.volatility,
            n_periods=params.years * 12,
            n_paths=10000
        )
        results.append(MonteCarloResult(
            mean_return=np.mean(paths[-1]),
            p10=np.percentile(paths[-1], 10),
            p90=np.percentile(paths[-1], 90),
            sharpe=params.expected_return / params.volatility
        ))
    return results

def variation_generator(base_params: dict, user_risk: str) -> List[dict]:
    """Generate ±15% variations based on risk tolerance"""
    variations = []
    risk_adjust = {"conservative": -0.02, "moderate": 0.0, "aggressive": 0.02}
    for adj in [-0.15, 0.0, 0.15]:
        var = base_params.copy()
        var["expected_return"] += risk_adjust[user_risk] + adj
        variations.append(var)
    return variations
